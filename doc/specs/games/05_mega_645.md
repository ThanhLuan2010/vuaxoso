```typescript
const winNums = new Set(draw.winningNumbers); // 6 số chính từ 01-45

for (const item of order.items) {
  const combos = item.playType === 'Cơ bản' 
    ? [item.numbers] 
    : getCombinationsList(item.numbers, 6);

  for (const combo of combos) {
    const matchCount = combo.filter(n => winNums.has(n)).length;

    if (matchCount === 6) {
      itemPrize += JACKPOT_VALUE;
    } else if (matchCount === 5) {
      itemPrize += 10000000;
    } else if (matchCount === 4) {
      itemPrize += 300000;
    } else if (matchCount === 3) {
      itemPrize += 30000;
    }
  }
}
```
